SELECT e1.*
FROM Employees e1
JOIN Employees e2
ON e1.salary < e2.salary
GROUP BY e1.emp_id, e1.salary
HAVING COUNT(DISTINCT e2.salary) = 2;

SELECT p1.name AS person1,
       p2.name AS person2,
       p1.address,
       p2.address
FROM People p1
JOIN People p2
ON SUBSTRING_INDEX(p1.name, ' ', -1) = SUBSTRING_INDEX(p2.name, ' ', -1)
AND p1.address <> p2.address
AND p1.id < p2.id;

SELECT e.emp_name, e.join_date,
       m.emp_name, m.join_date
FROM Employees e
JOIN Employees m
ON e.manager_id = m.emp_id;


SELECT e.emp_name AS employee,
       e.join_date AS emp_join_date,
       m.emp_name AS manager,
       m.join_date AS manager_join_date
FROM Employees e
JOIN Employees m
ON e.manager_id = m.emp_id
WHERE e.join_date < m.join_date;

SELECT l1.user_id,
       l1.login_date AS current_day,
       l2.login_date AS previous_day
FROM LoginLog l1
JOIN LoginLog l2
ON l1.user_id = l2.user_id
AND l1.login_date = DATE_ADD(l2.login_date, INTERVAL 1 DAY);
